// 3. ENVIAR NOTIFICAÇÕES DO BACKEND (Cloud Function)
// ============================================================
// Crie uma Cloud Function para enviar notificações:

const functions = require('firebase-functions');
const admin = require('firebase-admin');

admin.initializeApp();

exports.sendNotification = functions.https.onCall(async (data, context) => {
  // Verificar autenticação
  if (!context.auth) {
    throw new functions.https.HttpsError('unauthenticated', 'Usuário não autenticado');
  }

  const { userId, title, body, tag } = data;

  // Buscar todos os FCM tokens do usuário
  const tokensSnapshot = await admin
    .firestore()
    .collection(`users/${userId}/fcmTokens`)
    .get();

  const tokens = tokensSnapshot.docs.map(doc => doc.data().token);

  if (tokens.length === 0) {
    // Se não há tokens, apenas salvar no Firestore
    await admin.firestore()
      .collection(`users/${userId}/notifications`)
      .add({
        userId,
        title,
        body,
        tag,
        createdAt: admin.firestore.FieldValue.serverTimestamp(),
        read: false,
      });
    return { success: true, method: 'firestore' };
  }

  // Enviar via FCM
  const message = {
    notification: {
      title,
      body,
      icon: '/favicon.png',
    },
    data: {
      tag: tag || 'default',
    },
  };

  const response = await admin.messaging().sendMulticast({
    tokens,
    notification: message.notification,
    data: message.data,
    webpush: {
      notification: {
        title,
        body,
        icon: '/favicon.png',
        badge: '/favicon.png',
      },
    },
  });

  // Salvar notificação no Firestore também (para histórico)
  await admin.firestore()
    .collection(`users/${userId}/notifications`)
    .add({
      userId,
      title,
      body,
      tag,
      createdAt: admin.firestore.FieldValue.serverTimestamp(),
      read: false,
    });

  return {
    success: true,
    successCount: response.successCount,
    failureCount: response.failureCount,
    method: 'fcm',
  };
});

// 4. USAR NO SEU APP
// ============================================================
// Chamar a função do seu frontend quando necessário:

import { getFunctions, httpsCallable } from 'firebase/functions';

const functions = getFunctions();
const sendNotification = httpsCallable(functions, 'sendNotification');

// Exemplos de uso:
await sendNotification({
  userId: 'user-id',
  title: 'Nova mensagem',
  body: 'Você recebeu uma mensagem de João',
  tag: 'messages',
});
